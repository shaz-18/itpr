from PIL import Image, ImageOps

def color_negative(image_path, save_path):
    image = Image.open(image_path).convert("RGB")  # Ensure 24-bit RGB
    negative = ImageOps.invert(image)
    negative.save(save_path)
    print(f"24-bit negative saved to {save_path}")

def grayscale_negative(image_path, save_path):
    image = Image.open(image_path).convert("L")  # Convert to 8-bit grayscale
    negative = ImageOps.invert(image)
    negative.save(save_path)
    print(f"8-bit grayscale negative saved to {save_path}")

# Example usage:

Q3. (Example) Solve following with help of python library.1)Given an image with a point P(x,y)=(2,3), apply a translation of 4 units to the right and 5 units upwards. What will be the new coordinates of the point after translation?  
2) Given a point P(2,3), rotate it by 90 degrees counterclockwise around the origin. What are the new coordinates of the point after rotation?
3) Given an image point P(3,4), apply a scaling factor of 2 along the x-axis and 3 along the y-axis. What will be the new coordinates of the point after scaling?
4) Given a point P(2,3), apply a shear along the x-axis with a shear factor of 2. What are the new coordinates of the point after shearing?

import numpy as np

# Translation function
def translate(point, tx, ty):
    x, y = point
    return x + tx, y + ty

# Rotation function (90 degrees counterclockwise)
def rotate_90_ccw(point):
    x, y = point
    rotation_matrix = np.array([[0, -1], [1, 0]])
    new_x, new_y = np.dot(rotation_matrix, np.array([x, y]))
    return int(new_x), int(new_y)

# Scaling function
def scale(point, sx, sy):
    x, y = point
    return x * sx, y * sy

# Shearing function (along x-axis)
def shear_x(point, shear_factor):
    x, y = point
    new_x = x + shear_factor * y
    return new_x, y

# Given points and transformations
point1 = (2, 3)
translated_point = translate(point1, 4, 5)
print(f"Translated Point: {translated_point}")  # (6, 8)

rotated_point = rotate_90_ccw(point1)
print(f"Rotated Point: {rotated_point}")  # (-3, 2)

point2 = (3, 4)
scaled_point = scale(point2, 2, 3)
print(f"Scaled Point: {scaled_point}")  # (6, 12)

sheared_point = shear_x(point1, 2)
print(f"Sheared Point: {sheared_point}")  # (8, 3)

# Step function (activation)
def step_function(net_input):
    return 1 if net_input >= 0 else 0

# Perceptron training
def train_perceptron(inputs, targets, learning_rate=0.1, epochs=10):
    num_features = inputs.shape[1]
    weights = np.zeros(num_features)
    bias = 0

    for epoch in range(epochs):
        print(f"\nEpoch {epoch+1}")
        for i in range(len(inputs)):
            x = inputs[i]
            d = targets[i]
            net_input = np.dot(weights, x) + bias
            y = step_function(net_input)

            # Weight update rule
            error = d - y
            weights += learning_rate * error * x
            bias += learning_rate * error

            print(f"Input: {x}, Target: {d}, Output: {y}, Updated Weights: {weights}, Bias: {bias}")

    return weights, bias

    for i in range(len(inputs)):
        input_vector = inputs[i]
        desired_output = desired_outputs[i]
        
        # Calculate actual output using the weights
        actual_output = np.dot(input_vector, weights)

    print(f'Epoch {epoch+1}, Weights: {weights}')

4.1 Realization of logical AND function using Perceptron learning rule.

import numpy as np

# Define the Perceptron class
class Perceptron:
    def __init__(self, input_size, learning_rate=0.1, epochs=10):
        self.weights = np.zeros(input_size + 1)  # +1 for bias term
        self.learning_rate = learning_rate
        self.epochs = epochs

    def activation(self, x):
        return 1 if x >= 0 else 0

    def predict(self, inputs):
        summation = np.dot(inputs, self.weights[1:]) + self.weights[0]  # Bias is included in weights[0]
        return self.activation(summation)

    def train(self, training_inputs, labels):
        for _ in range(self.epochs):
            for inputs, label in zip(training_inputs, labels):
                prediction = self.predict(inputs)
                error = label - prediction
                # Update weights and bias using Perceptron learning rule
                self.weights[1:] += self.learning_rate * error * inputs
                self.weights[0] += self.learning_rate * error

# Define the training data for AND function
training_data_and = [
    (np.array([0, 0]), 0),
    (np.array([0, 1]), 0),
    (np.array([1, 0]), 0),
    (np.array([1, 1]), 1),
]

# Training the Perceptron for AND function
def train_perceptron_for_and():
    perceptron = Perceptron(input_size=2)
    inputs = [data[0] for data in training_data_and]
    labels = [data[1] for data in training_data_and]
    perceptron.train(inputs, labels)
    
    print("Trained weights for AND function:", perceptron.weights)
    return perceptron

and_perceptron = train_perceptron_for_and()

# Test the Perceptron for AND function
def test_perceptron_and(perceptron):
    test_data = np.array([[0, 0], [0, 1], [1, 0], [1, 1]])
    predictions = [perceptron.predict(data) for data in test_data]
    return predictions

print("\nTesting AND function predictions:")
and_predictions = test_perceptron_and(and_perceptron)
print(and_predictions)

              metrics=['accuracy'])

model.fit(train_images, train_labels, epochs=5)

# Evaluate the model on the test set

predicted_label = np.argmax(predictions[0])

import speech_recognition as sr

def recognize_speech():
    # Initialize recognizer
    recognizer = sr.Recognizer()

    # Use the microphone for input
    with sr.Microphone() as source:
        print("Adjusting for ambient noise...")
        recognizer.adjust_for_ambient_noise(source)  # Adjusts for ambient noise
        print("Listening for your speech...")
        audio = recognizer.listen(source)  # Listen to the speech

    try:
        # Use Google Web Speech API to recognize speech
        print("Recognizing speech...")
        text = recognizer.recognize_google(audio)  # Recognize the speech
        print(f"You said: {text}")  # Output the recognized text

    except sr.UnknownValueError:
        print("Sorry, I could not understand the audio.")  # If speech is not clear
    except sr.RequestError:
        print("Sorry, the service is down or there is a problem with the API.")  # API issues

if __name__ == "__main__":
    recognize_speech()

import yfinance as yf
import pandas as pd
import numpy as np
from sklearn.preprocessing import MinMaxScaler
from tensorflow.keras.models import Sequential
from tensorflow.keras.layers import Dense
import matplotlib.pyplot as plt

# 📌 Step 1: User Input
stock_symbol = input("Enter stock symbol (e.g., AAPL, MSFT): ")
start_date = input("Enter start date (YYYY-MM-DD): ")
end_date = input("Enter end date (YYYY-MM-DD): ")

# 📌 Step 2: Fetch Data
data = yf.download(stock_symbol, start=start_date, end=end_date)
if data.empty:
    print("No data found. Check your symbol and date range.")
    exit()

# 📌 Step 3: Use only 'Close' price
data = data[['Close']]

# 📌 Step 4: Preprocessing
scaler = MinMaxScaler()
scaled_data = scaler.fit_transform(data)

# Create features and labels
def create_dataset(dataset, time_step=10):
    X, y = [], []
    for i in range(len(dataset)-time_step):
        X.append(dataset[i:(i+time_step), 0])
        y.append(dataset[i + time_step, 0])
    return np.array(X), np.array(y)

X, y = create_dataset(scaled_data)

# Reshape for FFNN (not 3D like LSTM)
X_train = X
y_train = y

# 📌 Step 5: Build Feed Forward Neural Network
model = Sequential()
model.add(Dense(64, input_dim=X_train.shape[1], activation='relu'))
model.add(Dense(32, activation='relu'))
model.add(Dense(1))
model.compile(optimizer='adam', loss='mean_squared_error')

# 📌 Step 6: Train Model
model.fit(X_train, y_train, epochs=50, batch_size=32, verbose=0)

# 📌 Step 7: Predict and Plot
predicted = model.predict(X_train)
predicted_prices = scaler.inverse_transform(predicted)
real_prices = scaler.inverse_transform(y_train.reshape(-1, 1))

# Plot
plt.figure(figsize=(10, 5))
plt.plot(real_prices, label='Actual Price')
plt.plot(predicted_prices, label='Predicted Price')
plt.title(f'{stock_symbol} Stock Price Prediction')
plt.xlabel('Days')
plt.ylabel('Price')
plt.legend()
plt.show()
